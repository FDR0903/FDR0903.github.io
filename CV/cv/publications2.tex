% Nicely styled (clickable) link marker
\providecommand{\reflink}{{\color{teal}\underline{link}}}

% A numbered paper. #1 may contain a \\ separating the title line from the venue.
\providecommand{\cvpubn}[1]{%
  \par\refstepcounter{papercount}%
  {\parskip=0pt \leftskip=\papnumw \parindent=-\papnumw
   \fontsize{10pt}{1.12em}\bodyfont\color{text}%
   \makebox[\papnumw][l]{\color{gray}\bfseries\small\thepapercount.}#1\par}%
  \vspace{0.8mm}%
}

% "Presented at:" aligned under the paper text; first venue inline, rest beneath it
\providecommand{\presentedat}[1]{%
  \par\vspace{-0.8mm}%
  {\parskip=0pt \leftskip=\papnumw \parindent=0pt
   \fontsize{8.8pt}{10.2pt}\bodyfont\itshape\color{gray}%
   \settowidth{\presatw}{Presented at: }%
   \hangindent=\presatw \hangafter=1
   \noindent Presented at: #1\par}%
  \vspace{1mm}%
}
